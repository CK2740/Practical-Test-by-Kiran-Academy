-- 1. DATABASE SETUP
-- ============================================================

CREATE DATABASE IF NOT EXISTS streamflix;
USE streamflix;

DROP TABLE IF EXISTS movies;

CREATE TABLE movies (
    movie_id            INT PRIMARY KEY,
    title                VARCHAR(100),
    genre                VARCHAR(50),
    language             VARCHAR(30),
    rating               DOUBLE,
    price                DOUBLE,
    views                INT,
    release_year         INT,
    director             VARCHAR(100),
    production_company   VARCHAR(100)
);

-- Task 1: Movies with rating greater than 8
SELECT * FROM movies WHERE rating > 8;

-- Task 2: Movies whose price is less than 150
SELECT * FROM movies WHERE price < 150;

-- Task 3: All Hindi movies
SELECT * FROM movies WHERE language = 'Hindi';

-- Task 4: Movies with more than 1,000,000 views
SELECT * FROM movies WHERE views > 1000000;

-- Task 5: Movies released after 2015
SELECT * FROM movies WHERE release_year > 2015;

-- Task 6: Movies whose rating is between 8 and 8.5
SELECT * FROM movies WHERE rating BETWEEN 8 AND 8.5;

-- Task 7: Movies belonging to Action, Comedy or Thriller (IN)
SELECT * FROM movies WHERE genre IN ('Action', 'Comedy', 'Thriller');

-- Task 8: Movies directed by S. S. Rajamouli
SELECT * FROM movies WHERE director = 'S. S. Rajamouli';

-- Task 9: Movies produced by Hombale Films

-- Task 10: Movies produced by Arka Media Works

-- ============================================================
-- LEVEL 2 — STREAMING RECOMMENDATION TEAM
-- ============================================================

-- Task 11: 5 most viewed movies
SELECT * FROM movies ORDER BY views DESC LIMIT 5;

-- Task 12: 5 lowest-priced movies
SELECT * FROM movies ORDER BY price ASC LIMIT 5;

-- Task 13: Movies from highest rating to lowest rating
SELECT * FROM movies ORDER BY rating DESC;

-- Task 14: Unique movie languages (DISTINCT)
SELECT DISTINCT language FROM movies;

-- Task 15: Movie and Rating using aliases Movie and IMDb_Rating
SELECT title AS Movie, rating AS IMDb_Rating FROM movies;

-- Task 16: Movie title, director and production company only

-- Task 17: Sort by production company, then by rating descending
SELECT * FROM movies ORDER BY production_company ASC, rating DESC;

-- Task 18: 5 highest-rated movies released after 2015

-- Task 19: 3 most viewed Telugu movies

-- Task 20: 5 most expensive movies

-- ============================================================
-- LEVEL 5 — BOSS CHALLENGES
-- ============================================================

-- Challenge 1: Hidden Gem — rating above 8 but fewer than 500,000 views
SELECT * FROM movies WHERE rating > 8 AND views < 500000;

-- Challenge 2: Cheap & Popular — price below 150 and views above 1,000,000
SELECT * FROM movies WHERE price < 150 AND views > 1000000;

-- Challenge 3: Recommendation Engine — English movies released after 2015, rating above 8.5
SELECT * FROM movies WHERE language = 'English' AND release_year > 2015 AND rating > 8.5;

-- Challenge 4: Trending Movies — 3 most viewed movies released after 2020
SELECT * FROM movies WHERE release_year > 2020 ORDER BY views DESC LIMIT 3;

-- Challenge 5: Premium Movies — price greater than the average movie price (AVG)
SELECT * FROM movies WHERE price > (SELECT AVG(price) FROM movies);

-- Challenge 6: Director Spotlight — all S. S. Rajamouli movies, sorted by release year
SELECT * FROM movies WHERE director = 'S. S. Rajamouli' ORDER BY release_year ASC;

-- Challenge 7: Production House Report — Hombale Films: title, year, rating, views
SELECT title, release_year, rating, views FROM movies WHERE production_company = 'Hombale Films';

-- Challenge 8: Franchise Search — title contains Baahubali or KGF
SELECT * FROM movies WHERE title LIKE '%Baahubali%' OR title LIKE '%KGF%';

-- ============================================================
-- FINAL BOSS — CEO CHALLENGE
-- ============================================================
-- Best-performing movies report: title, genre, rating, price, views, director, production company
-- Conditions: rating > 8, views > 500000, price BETWEEN 100 AND 250
-- Sort by views DESC, show only top 5

SELECT title, genre, rating, price, views, director, production_company
FROM movies
WHERE rating > 8
  AND views > 500000
  AND price BETWEEN 100 AND 250
ORDER BY views DESC
LIMIT 5;
